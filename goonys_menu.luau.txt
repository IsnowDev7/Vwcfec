local ActiveHighlights = {} -- Instance -> Highlight

local function clearHighlights()
    for inst, hl in pairs(ActiveHighlights) do
        if hl and hl.Parent then
            hl:Destroy()
        end
    end
    table.clear(ActiveHighlights)
end

local function applyHighlight(inst, color, fillTrans, outlineTrans)
    if not inst or not inst.Parent then return end
    local existing = ActiveHighlights[inst] or inst:FindFirstChild("GoonysHighlight")
    if not existing then
        local hl = Instance.new("Highlight")
        hl.Name = "GoonysHighlight"
        hl.FillColor = color
        hl.OutlineColor = Color3.fromRGB(255, 255, 255)
        hl.FillTransparency = fillTrans or 0.4
        hl.OutlineTransparency = outlineTrans or 0.1
        hl.Adornee = inst
        hl.Parent = inst
        ActiveHighlights[inst] = hl
    else
        existing.FillColor = color
        existing.Enabled = true
        ActiveHighlights[inst] = existing
    end
end

local function removeHighlight(inst)
    local hl = ActiveHighlights[inst] or (inst and inst:FindFirstChild("GoonysHighlight"))
    if hl then
        hl:Destroy()
        ActiveHighlights[inst] = nil
    end
end

-- ESP Update Loop
local function updateESP()
    -- 1. Player ESP
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            if PlayerEspEnabled then
                applyHighlight(p.Character, Color3.fromRGB(255, 50, 50), 0.5, 0)
            else
                removeHighlight(p.Character)
            end
        end
    end

    -- 2. Egg ESP
    local stages = Workspace.Map:FindFirstChild("Stages")
    if stages then
        for _, stage in ipairs(stages:GetChildren()) do
            local spawned = stage:FindFirstChild("SpawnedEggs")
            if spawned then
                for _, egg in ipairs(spawned:GetChildren()) do
                    if EggEspEnabled then
                        applyHighlight(egg, Color3.fromRGB(255, 215, 0), 0.4, 0)
                    else
                        removeHighlight(egg)
                    end
                end
            end
        end
    end

    -- 3. Dino Bone / Fossil ESP
    if stages then
        for _, stage in ipairs(stages:GetChildren()) do
            local spawned = stage:FindFirstChild("SpawnedFossils")
            if spawned then
                for _, fossil in ipairs(spawned:GetChildren()) do
                    if BoneEspEnabled then
                        applyHighlight(fossil, Color3.fromRGB(0, 220, 255), 0.35, 0)
                    else
                        removeHighlight(fossil)
                    end
                end
            end
        end
    end

    local adminFossils = Workspace:FindFirstChild("AdminFossils")
    if adminFossils then
        for _, f in ipairs(adminFossils:GetChildren()) do
            if BoneEspEnabled then
                applyHighlight(f, Color3.fromRGB(0, 220, 255), 0.35, 0)
            else
                removeHighlight(f)
            end
        end
    end
end

        return true

-- ==================== SMOOTH FLIGHT & MOVEMENT LOGIC ====================
local Flying = false
local FlySpeed = 60
local BodyGyro, BodyVelocity
local FlyConnection

local function stopFlying()
    if FlyConnection then
        FlyConnection:Disconnect()
        FlyConnection = nil
    end
    if BodyGyro then
        BodyGyro:Destroy()
        BodyGyro = nil
    end
    if BodyVelocity then
        BodyVelocity:Destroy()
        BodyVelocity = nil
    end

    local character = LocalPlayer.Character
    if character then
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if humanoid then
            humanoid.PlatformStand = false
        end
    end
end

local function startFlying()
    stopFlying()
    local character = LocalPlayer.Character
    if not character then return end
    local rootPart = character:FindFirstChild("HumanoidRootPart")
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not rootPart or not humanoid then return end

    humanoid.PlatformStand = true

    BodyGyro = Instance.new("BodyGyro")
    BodyGyro.P = 150000
    BodyGyro.D = 800
    BodyGyro.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    BodyGyro.CFrame = rootPart.CFrame
    BodyGyro.Parent = rootPart

    BodyVelocity = Instance.new("BodyVelocity")
    BodyVelocity.Velocity = Vector3.zero
    BodyVelocity.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    BodyVelocity.Parent = rootPart

    local currentVelocity = Vector3.zero

    FlyConnection = RunService.RenderStepped:Connect(function(deltaTime)
        if not Flying or not character.Parent or not rootPart.Parent or not humanoid.Parent then
            stopFlying()
            return
        end

        local camCFrame = Camera.CFrame
        local targetCFrame = CFrame.new(rootPart.Position, rootPart.Position + camCFrame.LookVector)
        BodyGyro.CFrame = BodyGyro.CFrame:Lerp(targetCFrame, math.clamp(deltaTime * 18, 0, 1))

        local rawDirection = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then
            rawDirection = rawDirection + camCFrame.LookVector
        end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then
            rawDirection = rawDirection - camCFrame.LookVector
        end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then
            rawDirection = rawDirection - camCFrame.RightVector
        end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then
            rawDirection = rawDirection + camCFrame.RightVector
        end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
            rawDirection = rawDirection + Vector3.new(0, 1, 0)
        end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then
            rawDirection = rawDirection - Vector3.new(0, 1, 0)
        end

        local targetVelocity = Vector3.zero
        if rawDirection.Magnitude > 0 then
            targetVelocity = rawDirection.Unit * FlySpeed
        end

        currentVelocity = currentVelocity:Lerp(targetVelocity, math.clamp(deltaTime * 12, 0, 1))
        BodyVelocity.Velocity = currentVelocity
    end)
end

local function applySpeedAndJump()
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if hum then
        if SpeedEnabled then
            hum.WalkSpeed = BaseWalkSpeed * SpeedMultiplier
        else
            hum.WalkSpeed = BaseWalkSpeed
        end

        if JumpEnabled then
            hum.UseJumpPower = true
            hum.JumpPower = BaseJumpPower * JumpMultiplier
        else
            hum.UseJumpPower = true
            hum.JumpPower = BaseJumpPower
        end
    end
end

local movementStep = RunService.Heartbeat:Connect(function()
    if not Flying then
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hum and hum.Health > 0 then
            local desiredSpeed = SpeedEnabled and (BaseWalkSpeed * SpeedMultiplier) or BaseWalkSpeed
            local desiredJump = JumpEnabled and (BaseJumpPower * JumpMultiplier) or BaseJumpPower

            if hum.WalkSpeed ~= desiredSpeed then
                hum.WalkSpeed = desiredSpeed
            end
            if hum.JumpPower ~= desiredJump then
                hum.UseJumpPower = true
                hum.JumpPower = desiredJump
            end
        end
    end
end)

local function getPlayerPlot()
    local plots = Workspace.Map:FindFirstChild("Plots")
    if plots then
        for _, plot in ipairs(plots:GetChildren()) do
            local owner = plot:FindFirstChild("Owner")
            if owner and (tostring(owner.Value) == LocalPlayer.Name or tostring(owner.Value) == tostring(LocalPlayer.UserId)) then
                return plot
            end
        end
    end
    return nil
end

local function getPlayerBaseCFrame()
    local plot = getPlayerPlot()
    if plot then
        local zone = plot:FindFirstChild("SquatZone") or plot:FindFirstChild("DefaultSize") or plot:FindFirstChild("Detector")
        if zone then
            local cf = zone:IsA("Model") and zone:GetPivot() or zone.CFrame
            return cf + Vector3.new(0, 3, 0)
        end
        return plot:GetPivot() + Vector3.new(0, 3, 0)
    end
    return nil
end

local function teleportToBase()
    local baseCF = getPlayerBaseCFrame()
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if root and baseCF then
        instantTeleport(baseCF)
        if ReturnToBaseRemote then
            pcall(function() ReturnToBaseRemote:FireServer() end)
        end
        return true
    end
    return false
end

-- ==================== EGG HARVESTING ENGINE ====================
local function isEggAllowed(eggName)
    local hasSpecificEgg = false
    for _, isSel in pairs(SelectedEggs) do
        if isSel then
            hasSpecificEgg = true
            break
        end
    end

    if not hasSpecificEgg then
        return true
    end

    local lowerName = string.lower(eggName)
    for sel, isSel in pairs(SelectedEggs) do
        if isSel and (string.lower(sel) == lowerName or string.find(lowerName, string.lower(sel))) then
            return true
        end
    end

    return false
end

local function isZoneAllowed(zoneName)
    local hasSpecificZone = false
    for _, isSel in pairs(SelectedZones) do
        if isSel then
            hasSpecificZone = true
            break
        end
    end

    if not hasSpecificZone then
        return true
    end

    return SelectedZones[zoneName] == true
end

local function getTargetEggPrompt()
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return nil, nil end

    local closestPrompt = nil
    local closestPart = nil
    local shortestDist = math.huge

    local stagesFolder = Workspace.Map:FindFirstChild("Stages")
    if stagesFolder then
        for _, stage in ipairs(stagesFolder:GetChildren()) do
            if isZoneAllowed(stage.Name) then
                local spawned = stage:FindFirstChild("SpawnedEggs")
                if spawned then
                    for _, egg in ipairs(spawned:GetChildren()) do
                        if isEggAllowed(egg.Name) then
                            local prompt = egg:FindFirstChild("CollectPrompt", true)
                            if prompt and prompt.Enabled then
                                local part = egg.PrimaryPart or egg:FindFirstChild("EggRoot") or egg:FindFirstChildWhichIsA("BasePart", true)
                                if part then
                                    local dist = (root.Position - part.Position).Magnitude
                                    if dist < shortestDist then
                                        shortestDist = dist
                                        closestPrompt = prompt
                                        closestPart = part
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
    end

    if TargetExclusives or not next(SelectedZones) then
        for _, obj in ipairs(Workspace:GetChildren()) do
            if obj.Name ~= "Terrain" and obj.Name ~= "Map" and obj.Name ~= "Camera" then
                local prompt = obj:FindFirstChild("CollectPrompt", true)
                if prompt and prompt.Enabled and isEggAllowed(obj.Name) then
                    local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart", true)
                    if part then
                        local dist = (root.Position - part.Position).Magnitude
                        if dist < shortestDist then
                            shortestDist = dist
                            closestPrompt = prompt
                            closestPart = part
                        end
                    end
                end
            end
        end

        for _, mapChild in ipairs(Workspace.Map:GetChildren()) do
            if mapChild.Name ~= "Stages" and mapChild.Name ~= "Plots" then
                local prompt = mapChild:FindFirstChild("CollectPrompt", true)
                if prompt and prompt.Enabled and isEggAllowed(mapChild.Name) then
                    local part = mapChild:IsA("BasePart") and mapChild or mapChild:FindFirstChildWhichIsA("BasePart", true)
                    if part then
                        local dist = (root.Position - part.Position).Magnitude
                        if dist < shortestDist then
                            shortestDist = dist
                            closestPrompt = prompt
                            closestPart = part
                        end
                    end
                end
            end
        end
    end

    return closestPrompt, closestPart
end

local function runEggLoop()
    while AutoCollectEggs do
        local char = LocalPlayer.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if not root then
            task.wait(1)
            continue
        end

        local prompt, part = getTargetEggPrompt()
        if prompt and part then
            instantTeleport(part.CFrame + Vector3.new(0, 2, 0))
            task.wait(0.15)

            triggerPrompt(prompt)
            task.wait(0.35)

            teleportToBase()
            task.wait(0.6)
        else
            task.wait(1)
        end
    end
end

local function isHoldingOrPendingFossil()

    local pendingName = LocalPlayer:GetAttribute("PendingFossilName")
    if carried > 0 or (typeof(pendingName) == "string" and pendingName ~= "") then

    return false
end

local FossilRarityTier = {
    ["Mythical"] = 6000,
    ["Ancient"]  = 5000,
    ["Rare"]     = 4000,
    ["Large"]    = 3000,
    ["Common"]   = 2000,
    ["Small"]    = 1000
}

local function getFossilScore(fossilModel, dist)
    local sizeAttr = fossilModel:GetAttribute("FossilSize") or ""
    local fragmentsAttr = tonumber(fossilModel:GetAttribute("Fragments")) or 0
    local tierValue = FossilRarityTier[sizeAttr] or 500

    -- High tier/fragments gets much higher score; subtract distance factor so closest among same tier wins
    return (tierValue * 10) + (fragmentsAttr * 100) - (dist * 0.05)
end

local function getAvailableFossil()
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return nil, nil end

    local stagesFolder = Workspace.Map:FindFirstChild("Stages")
    if not stagesFolder then return nil, nil end

    local candidatePrompt = nil
    local candidatePart = nil
    local bestScore = -math.huge
    local shortestDist = math.huge

    local function checkFossil(fossilModel)
        -- Skip if someone else is already carrying it
        local carriedBy = fossilModel:GetAttribute("CarriedByUserId")
        if carriedBy and carriedBy ~= 0 and carriedBy ~= LocalPlayer.UserId then
            return
        end

        local prompt = fossilModel:FindFirstChild("DigPrompt", true) or fossilModel:FindFirstChildWhichIsA("ProximityPrompt", true)
        if prompt and prompt.Enabled then
            local part = fossilModel:IsA("BasePart") and fossilModel or fossilModel:FindFirstChildWhichIsA("BasePart", true)
            if part then
                local dist = (root.Position - part.Position).Magnitude
                if PrioritizeBetterBones then
                    local score = getFossilScore(fossilModel, dist)
                    if score > bestScore then
                        bestScore = score
                        candidatePrompt = prompt
                        candidatePart = part
                    end
                else
                    if dist < shortestDist then
                        shortestDist = dist
                        candidatePrompt = prompt
                        candidatePart = part
                    end
                end
            end
        end
    end

    for _, stage in ipairs(stagesFolder:GetChildren()) do
        local spawned = stage:FindFirstChild("SpawnedFossils")
        if spawned then
            for _, fossilModel in ipairs(spawned:GetChildren()) do
                checkFossil(fossilModel)
            end
        end
    end

    local adminFossils = Workspace:FindFirstChild("AdminFossils")
    if adminFossils then
        for _, f in ipairs(adminFossils:GetChildren()) do
            checkFossil(f)
        end
    end

    return candidatePrompt, candidatePart
end

local function instantCleanBone()
    local tablePrompt, tableDetector = getTableInfo()
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root or not tableDetector then return end

    instantTeleport(tableDetector.CFrame + Vector3.new(0, 3, 0))
    task.wait(0.2)

    if tablePrompt and tablePrompt.Enabled then
        triggerPrompt(tablePrompt)
        task.wait(0.25)
    end

    if FossilRemote then
        pcall(function()
            FossilRemote:FireServer("Finish")
        end)
    end

    local maxWait = 3.5
    local waited = 0
    while isHoldingOrPendingFossil() and waited < maxWait do
        task.wait(0.2)
        waited = waited + 0.2
        if FossilRemote then
            pcall(function() FossilRemote:FireServer("Finish") end)
        end
    end
    task.wait(0.25)
end

local function runFossilLoop()
    while AutoFarmFossils do
        local char = LocalPlayer.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if not root then
            task.wait(1)
            continue
        end

        if isHoldingOrPendingFossil() then
            if AutoCleanBones then
                instantCleanBone()
            end
            task.wait(0.3)
            continue
        end

        local prompt, part = getAvailableFossil()
        if prompt and part then
            instantTeleport(part.CFrame + Vector3.new(0, 2, 0))
            task.wait(0.2)

            triggerPrompt(prompt)

            local pickupWait = 0
            while not isHoldingOrPendingFossil() and pickupWait < 1.5 do
                task.wait(0.1)
                pickupWait = pickupWait + 0.1
            end
            task.wait(0.15)

            if AutoCleanBones and isHoldingOrPendingFossil() then
                instantCleanBone()
            end

            task.wait(0.4)
        else
            task.wait(1.5)
        end
    end
end

            else
                warn("[ServerHop] No suitable server found")
            end
        end)
    end
})

        local newKey = Enum.KeyCode[keyName]
        if newKey then

            if Window.SetToggleKey then
                pcall(function() Window:SetToggleKey(newKey) end)
            end
        end

local isMenuOpen = true
local lastToggleTick = 0

local function toggleMenuVisibility()
    local now = os.clock()
    if now - lastToggleTick < 0.35 then return end
    lastToggleTick = now
    if isMenuOpen then
        if Window.Close then pcall(function() Window:Close() end)
        elseif Window.Toggle then pcall(function() Window:Toggle() end) end
        isMenuOpen = false
    else
        if Window.Open then pcall(function() Window:Open() end)
        elseif Window.Toggle then pcall(function() Window:Toggle() end) end
        isMenuOpen = true
    end
end

KeybindConnection = UserInputService.InputBegan:Connect(function(input, processed)
    if not processed and input.KeyCode == MenuToggleKey then
        toggleMenuVisibility()
    end
end)

ExtrasTab:Button({
    Title = "Re-open / Toggle UI",
    Desc = "Manually toggle menu visibility",
    Callback = function()
        toggleMenuVisibility()
    end
})

local dragging = false
local dragInput, dragStart, startPos
local wasDragged = false

toggleButton.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        wasDragged = false
        dragStart = input.Position
        startPos = toggleButton.Position

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

toggleButton.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        if delta.Magnitude > 4 then
            wasDragged = true
        end
        toggleButton.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end
end)

toggleButton.MouseButton1Click:Connect(function()
    if wasDragged then
        wasDragged = false
        return
    end
    toggleMenuVisibility()
end)
